\section*{LAMPIRAN A TERM OF REFERENCE (TOR)}
\addcontentsline{toc}{chapter}{Lampiran A. Term of Reference (TOR)}

\begin{framed}
\begin{center}
{\bfseries\Large TERM OF REFERENCE (TOR)\par}
{\bfseries\large PENGEMBANGAN FUNGSI MANAJEMEN PENAGIHAN PIUTANG DAN ALUR REVISI PEMBAYARAN PARAPAY\par}
\vspace{0.2cm}
\textbf{PT PARAGON TECHNOLOGY AND INNOVATION}
\end{center}
\vspace{0.3cm}

\textit{Term of Reference} (TOR) ini merupakan dokumen acuan kerja yang menetapkan tujuan, batasan, metodologi, dan pembagian tanggung jawab dalam pelaksanaan Kerja Praktik pengembangan aplikasi Parapay di PT Paragon Technology and Innovation.

\begin{enumerate}[label=\arabic*.,leftmargin=0.8cm,itemsep=8pt,topsep=4pt,parsep=0pt,partopsep=0pt]
\item \textbf{TUJUAN PEKERJAAN}\par\vspace{2pt}
  Mengembangkan, merefaktor, dan mengoptimalkan fungsi aplikasi Parapay (sistem manajemen dan penagihan piutang usaha / \textit{Account Receivable}) berbasis JavaScript/TypeScript yang terintegrasi dua arah dengan SAP ERP. Tujuan utama pekerjaan meliputi implementasi alur simpan sementara (\textit{Save as Draft}), perancangan ulang arsitektur revisi pembayaran (Revision v2) dengan pemuatan otomatis data lama, integrasi penyimpanan bukti transfer luring berbasis \textit{IndexedDB}, serta dukungan stabilitas operasional di 28 pusat distribusi (\textit{Distribution Center / Regional Distribution Center} atau DC/RDC) nasional.

\item \textbf{LINGKUP PEKERJAAN}
  \begin{enumerate}[label=\alph*),topsep=2pt,itemsep=2pt,parsep=0pt,partopsep=0pt,leftmargin=0.5cm]
    \item Pengembangan antarmuka dan logika \textit{Save as Draft} pada alur pembuatan koleksi penagihan.
    \item Rekonsiliasi 7 tipe pembayaran (Tunai, Transfer Bank, Bilyet Giro, Potong Tagihan, Tukar Faktur, Retur, dan Nota Debit/Kredit).
    \item Pembuatan modul Potong Tagihan dengan indikator status dinamik dan pengendalian hak akses (\textit{Role-Based Access Control} / RBAC).
    \item Refactoring arsitektur Revision v1 menjadi Revision v2 yang mendukung pemuatan otomatis data lama, \textit{state invalidation}, mode \textit{read-only}, \textit{non-allocation lock}, serta alur khusus setor tunai (\textit{Cash Before Setor} dan \textit{Cash After Setor}).
    \item Integrasi API peramban \textit{IndexedDB} untuk penyimpanan bukti transfer secara luring di sisi klien.
    \item Optimasi antarmuka melalui \textit{server-side search} dan \textit{infinite scroll} pada penarikan data area dan pelanggan berskala besar (>50 entri).
    \item Penanganan tiket bantuan operasional harian dan eksekusi skrip perbaikan data (\textit{data patch}) pada basis data produksi.
  \end{enumerate}

\item \textbf{HASIL PEKERJAAN}
  \begin{enumerate}[label=\alph*),topsep=2pt,itemsep=2pt,parsep=0pt,partopsep=0pt,leftmargin=0.5cm]
    \item Modul aplikasi Parapay yang teruji, terintegrasi, dan berhasil digabungkan (\textit{merged}) ke dalam repositori utama PT Paragon Technology and Innovation.
    \item Penutupan tiket bantuan operasional dan pemulihan integritas data transaksi pada 28 DC/RDC.
    \item Laporan Kerja Praktik dan dokumentasi teknis sistem.
  \end{enumerate}

\item \textbf{BATASAN}
  \begin{enumerate}[label=\alph*),topsep=2pt,itemsep=2pt,parsep=0pt,partopsep=0pt,leftmargin=0.5cm]
    \item \textbf{Batasan Bisnis}: Dalam satu koleksi penagihan hanya boleh terdapat tepat 1 entri pembayaran tunai. Pembayaran tunai wajib melalui proses Setor Tunai sebelum dapat disetujui Admin AR.
    \item \textbf{Batasan Integrasi}: Master data faktur dan pencatatan buku besar akhir dikontrol secara terpusat oleh SAP ERP.
    \item \textbf{Batasan Lingkungan}: Pengembangan antarmuka difokuskan pada \textit{framework} React dan TypeScript, pengujian dilakukan pada lingkungan \textit{staging}, dan perbaikan data produksi dilakukan bersama pembimbing teknis.
  \end{enumerate}

\item \textbf{ASUMSI}
  \begin{enumerate}[label=\alph*),topsep=2pt,itemsep=2pt,parsep=0pt,partopsep=0pt,leftmargin=0.5cm]
    \item Ketersediaan akses jaringan internal ParagonCorp via VPN dan akses repositori GitLab perusahaan.
    \item Ketersediaan \textit{endpoint} API SAP ERP dan basis data \textit{staging} yang stabil untuk pengujian integrasi.
    \item Ketersediaan kunci akses model kecerdasan buatan (DeepSeek API via OpenCode) untuk mendukung alur kerja \textit{AI-Native SDLC}.
  \end{enumerate}

\item \textbf{METODOLOGI PEKERJAAN}
  \begin{enumerate}[label=\roman*.,topsep=2pt,itemsep=2pt,parsep=0pt,partopsep=0pt,leftmargin=0.5cm]
    \item \textbf{Penjelasan Metodologi}: Menggunakan pendekatan \textit{Agile Software Development} dengan koordinasi harian (\textit{Daily Standup}), peninjauan kode (\textit{Merge Request}), dan rilis bertahap.
    \item \textbf{Pemetaan Tahap Pelaksanaan}:
    \begin{itemize}[topsep=2pt,itemsep=2pt,parsep=0pt,partopsep=0pt,leftmargin=0.4cm]
      \item \textit{Tahap Inisiasi} (Minggu 1): Onboarding HR \& tim FATL, setup lingkungan pengembang, VPN, dan analisis sistem Parapay.
      \item \textit{Tahap Pengembangan Modul Awal} (Minggu 2--4): Fitur \textit{Save as Draft}, validasi tipe pembayaran FAR, dan modul Potong Tagihan.
      \item \textit{Tahap Refactoring Arsitektur} (Minggu 5--7): Pembangunan Revision v2, \textit{non-allocation lock}, alur setor tunai, dan pemisahan \textit{endpoint} revisi.
      \item \textit{Tahap Optimasi dan Luring} (Minggu 8): Integrasi \textit{IndexedDB} dan optimasi \textit{infinite scroll} / \textit{server-side search}.
      \item \textit{Tahap Dukungan Operasional \& Dokumentasi} (Minggu 9--10): Penanganan tiket operasional DC/RDC, \textit{data patch} SQL, dan penyusunan laporan.
    \end{itemize}
  \end{enumerate}

\item \textbf{LINGKUNGAN PENGEMBANGAN}
  \begin{itemize}[topsep=2pt,itemsep=2pt,parsep=0pt,partopsep=0pt,leftmargin=0.5cm]
    \item \textbf{Perangkat Keras}: Laptop pengembang (Arsitektur x86\_64, RAM 16 GB).
    \item \textbf{Perangkat Lunak}: Node.js, React, TypeScript, TanStack Query, Zustand, Sequelize, Git, OpenCode, Zed / VS Code, Zathura.
    \item \textbf{Jaringan}: Paragon Internal Network via VPN.
  \end{itemize}

\item \textbf{MEKANISME PENGENDALIAN DAN PEMANTAUAN}
  \begin{enumerate}[label=\alph*),topsep=2pt,itemsep=2pt,parsep=0pt,partopsep=0pt,leftmargin=0.5cm]
    \item \textbf{Pertemuan Harian (\textit{Daily Standup})}: Pelaporan kemajuan harian, sinkronisasi tugas, dan identifikasi kendala (\textit{blocker}).
    \item \textbf{Pengulasan Kode (\textit{Code Review})}: Setiap pembaruan kode wajib melalui peninjauan via GitLab \textit{Merge Request} sebelum digabungkan ke \textit{branch} utama.
    \item \textbf{Pencatatan Aktivitas}: Seluruh rincian pekerjaan dicatat pada log aktivitas harian (\textit{Daily Log Activity}).

\item \textbf{MANAJEMEN RISIKO}
  \begin{enumerate}[label=\alph*),topsep=2pt,itemsep=2pt,parsep=0pt,partopsep=0pt,leftmargin=0.5cm]
    \item \textbf{Risiko Minimnya Dokumentasi Sistem Lama}: Diminimalkan dengan proaktif mengonfirmasi batasan bisnis harian kepada pembimbing teknis dan tim SAP.
    \item \textbf{Risiko Ketergantungan API \textit{Backend}}: Diatasi dengan pembuatan \textit{mocking API} di sisi klien selama endpoint dalam proses pengerjaan.
    \item \textbf{Risiko Regresi Kode pada \textit{AI-Native SDLC}}: Diatasi dengan pemecahan komponen menjadi modular dan pemeriksaan teliti terhadap \textit{git diff} sebelum komit.
    \item \textbf{Risiko Anomali Basis Data Produksi}: Diatasi dengan eksekusi skrip \textit{data patch} SQL/Sequelize di bawah pengawasan langsung pembimbing teknis.
  \end{enumerate}
